42template <CrcRegister T>
43[[nodiscard]] constexpr T ReflectBits(T value, uint32_t bits) noexcept {
44 auto in = static_cast<uint32_t>(value);
45 uint32_t out = 0U;
46 for (uint32_t i = 0U; i < bits; ++i) {
47 out = (out << 1U) | (in & 1U);
48 in >>= 1U;
49 }
50 return static_cast<T>(out);

66template <CrcRegister T, T Poly, bool Reflected>
67[[nodiscard]] constexpr std::array<T, 256> MakeCrcTable() noexcept {
68 constexpr auto kWidth = static_cast<uint32_t>(std::numeric_limits<T>::digits);
69 constexpr uint32_t kTop = 1UL << (kWidth - 1U);
70 constexpr auto kMask = static_cast<uint32_t>(std::numeric_limits<T>::max());
71 constexpr auto kPoly = static_cast<uint32_t>(Poly);
72 constexpr auto kReflectedPoly = static_cast<uint32_t>(ReflectBits<T>(Poly, kWidth));
73 std::array<T, 256> table{};
74 for (uint32_t byte = 0U; byte < 256U; ++byte) {
75 uint32_t reg = 0U;
76 if constexpr (Reflected) {
77 reg = byte;
78 for (uint32_t bit = 0U; bit < 8U; ++bit) {
79 reg = ((reg & 1U) != 0U) ? ((reg >> 1U) ^ kReflectedPoly) : (reg >> 1U);
80 }
81 } else {
82 reg = byte << (kWidth - 8U);
83 for (uint32_t bit = 0U; bit < 8U; ++bit) {
84 reg = ((reg & kTop) != 0U) ? (((reg << 1U) ^ kPoly) & kMask) : ((reg << 1U) & kMask);
85 }
86 }
87 table[byte] = static_cast<T>(reg);
88 }
89 return table;
90}

102template <CrcRegister T, T Poly, T Init, bool Reflected, T XorOut>
103class Crc {
104public:
106 constexpr Crc() noexcept = default;
107
109 constexpr void Update(std::span<const uint8_t> data) noexcept {
110 for (const uint8_t byte : data) {
111 reg_ = step(reg_, byte);
112 }
113 }
114
116 [[nodiscard]] constexpr T Value() const noexcept {
117 return static_cast<T>(reg_ ^ static_cast<uint32_t>(XorOut));
118 }
119
121 constexpr void Reset() noexcept {
122 reg_ = kStart;
123 }
124
126 [[nodiscard]] static constexpr T Compute(std::span<const uint8_t> data) noexcept {
127 Crc crc{};
128 crc.Update(data);
129 return crc.Value();
130 }
131
132private:
133 static constexpr uint32_t kWidth = detail::WidthOf<T>();
134 static constexpr uint32_t kMask = detail::MaskOf<T>();
135 static constexpr std::array<T, 256> kTable = detail::MakeCrcTable<T, Poly, Reflected>();
136 static constexpr uint32_t kStart =
137 Reflected ? static_cast<uint32_t>(detail::ReflectBits<T>(Init, kWidth))
138 : static_cast<uint32_t>(Init);
139
140 // One table step. For an 8-bit register the shifted-out part is zero, so the same form serves
141 // every width.
142 [[nodiscard]] static constexpr uint32_t step(uint32_t reg, uint8_t byte) noexcept {
143 const auto in = static_cast<uint32_t>(byte);
144 uint32_t next = 0U;
145 if constexpr (Reflected) {
146 const uint32_t index = (reg ^ in) & 0xFFU;
147 next = (reg >> 8U) ^ static_cast<uint32_t>(kTable[index]);
148 } else {
149 const uint32_t index = ((reg >> (kWidth - 8U)) ^ in) & 0xFFU;
150 next = ((reg << 8U) & kMask) ^ static_cast<uint32_t>(kTable[index]);
151 }
152 return next;
153 }
154
155 uint32_t reg_{kStart};
156};
