104[[nodiscard]] constexpr Microseconds ToMicroseconds(Milliseconds ms) noexcept {
105 constexpr uint32_t kMax = std::numeric_limits<uint32_t>::max();
106 constexpr uint32_t kPerMs = 1000U;
107 const uint32_t count = (ms.Count() > (kMax / kPerMs)) ? kMax : (ms.Count() * kPerMs);
108 return Microseconds{count};
109}
