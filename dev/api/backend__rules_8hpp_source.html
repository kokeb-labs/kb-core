70template <typename Clock, typename Nap, typename Done>
71[[nodiscard]] bool PollUntil(Clock clock, Nap nap, Done done, uint64_t limit_us) noexcept {
72 uint64_t last = clock();
73 uint64_t spent = 0U;
74 bool held = done();
75 while ((!held) && (spent < limit_us)) {
76 nap();
77 const uint64_t now = clock();
78 spent += ForwardUs(now, last);
79 last = now;
80 held = done();
