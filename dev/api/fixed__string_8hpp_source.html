79 [[nodiscard]] constexpr bool AppendUnsigned(uint64_t value, uint32_t base = 10U,
80 std::size_t min_digits = 1U) noexcept {
81 bool ok = (base >= 2U) && (base <= 16U);
82 if (ok) {
83 std::array<char, detail::kMaxDigits> digits{};
84 std::size_t count = 0U;
85 uint64_t rest = value;
86 const std::size_t wanted =
87 (min_digits < detail::kMaxDigits) ? min_digits : detail::kMaxDigits;
88 do {
89 digits[count] = detail::kDigits[static_cast<std::size_t>(rest % base)];
90 rest /= base;
91 ++count;
92 } while ((rest != 0U) || (count < wanted));
93 // The digits are least significant first: append them in reverse.
94 std::array<char, detail::kMaxDigits> ordered{};
95 for (std::size_t i = 0U; i < count; ++i) {
96 ordered[i] = digits[count - 1U - i];
97 }
98 ok = Append(std::string_view{ordered.data(), count});
99 }
100 return ok;
101 }
