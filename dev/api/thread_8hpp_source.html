116 [[nodiscard]] kb::Result<void> Start(const ThreadConfig& config) noexcept {
117 static_assert(ThreadBody<Derived>,
118 "Derived must provide Setup, Step and Cleanup, each noexcept");
119 kb::Status status = kb::Status::AlreadyStarted;
120 if (state_.load(std::memory_order_acquire) != static_cast<uint8_t>(ThreadState::Idle)) {
121 status = kb::Status::AlreadyStarted;
122 } else if (config.name == nullptr) {
123 status = kb::Status::InvalidArgument;
124 } else if ((!wake_.IsCreated()) && (!wake_.Create(1U, 0U).IsOk())) {
125 status = kb::Status::Fault;
126 } else {
127 period_ = config.period;
128 stop_.store(false, std::memory_order_release);
129 state_.store(static_cast<uint8_t>(ThreadState::Starting), std::memory_order_release);
130 detail::ThreadParams params{};
131 params.core = this;
132 params.stack = stack_.data();
133 params.stack_bytes = StackBytes;
134 params.name = config.name;
135 params.priority = config.priority;
136 status = backend::ThreadCreate(storage_, params);
137 if (status != kb::Status::Ok) {
138 state_.store(static_cast<uint8_t>(ThreadState::Idle), std::memory_order_release);
139 }
140 }
141 return kb::Result<void>::FromStatus(status);
142 }

242 // Tests `done` once a tick until it holds or `timeout` has passed on the monotonic clock, so the
243 // wait ends within a tick of the timeout at any tick rate. True when `done` held.
244 template <typename Done>
245 [[nodiscard]] static bool pollUntil(Done done, Milliseconds timeout) noexcept {
246 constexpr uint64_t kUsPerMs = 1000U;
247 return detail::PollUntil([]() noexcept { return backend::Now().Us(); },
248 []() noexcept { backend::SleepTicks(1U); }, done,
249 static_cast<uint64_t>(timeout.Count()) * kUsPerMs);
250 }

258 static void runCore(detail::ThreadCore& core) noexcept {
259 static_cast<Thread&>(core).run();

262 void run() noexcept {
263 auto& body = static_cast<Derived&>(*this);
264 ThreadState end = ThreadState::Failed;
265 if (body.Setup()) {
266 state_.store(static_cast<uint8_t>(ThreadState::Running), std::memory_order_release);
267 loop(body);
268 end = ThreadState::Stopped;
269 }
270 body.Cleanup();
271 state_.store(static_cast<uint8_t>(end), std::memory_order_release);
272 backend::ThreadExit(storage_);
273 }
274
275 void loop(Derived& body) noexcept {
276 Periodic grid{};
277 grid.Reset();
278 bool more = true;
279 while (more && (!StopRequested())) {
280 const kb::Instant before = backend::Now();
281 more = body.Step();
282 record(kb::utils::ElapsedUs(backend::Now(), before));
283 if (more && (!StopRequested())) {
284 if (park_.load(std::memory_order_acquire)) {
285 park();
286 grid.Reset();
287 } else {
288 pace(grid);
289 }
290 }
291 }
292 }
293
294 // Blocks on the wake semaphore until Resume() or RequestStop(); a stale give only re-tests.
295 void park() noexcept {
296 parked_.store(true, std::memory_order_release);
297 while (park_.load(std::memory_order_acquire) && (!StopRequested())) {
298 static_cast<void>(wake_.TakeFor(kb::kMaxMilliseconds));
299 }
300 parked_.store(false, std::memory_order_release);
301 }
302
303 void pace(Periodic& grid) noexcept {
304 if (period_.Count() == 0U) {
305 backend::Yield();
306 } else if (!grid.Wait(period_)) {
307 static_cast<void>(overruns_.fetch_add(1U, std::memory_order_relaxed));
308 } else {
309 // On time: nothing to record.
310 }
311 }

313 void record(uint64_t step_us) noexcept {
314 constexpr uint64_t kMax = std::numeric_limits<uint32_t>::max();
315 const auto clipped = static_cast<uint32_t>(std::min(step_us, kMax));
316 if (clipped > max_step_us_.load(std::memory_order_relaxed)) {
317 max_step_us_.store(clipped, std::memory_order_relaxed);
318 }
319 static_cast<void>(steps_.fetch_add(1U, std::memory_order_relaxed));
320 }
