121[[nodiscard]] constexpr FaultRecordBytes EncodeRecord(const FaultRecord& record) noexcept {
122 FaultRecordBytes b{};
123 const std::span<uint8_t, kFaultRecordBytes> out{b};
124 StoreLe16(detail::kJournalMagic, out.first<2U>());
125 b[detail::kAtVersion] = detail::kJournalVersion;
126 b[detail::kAtKind] = static_cast<uint8_t>(record.kind);
127 StoreLe32(record.sequence, out.subspan<detail::kAtSequence, 4U>());
128 StoreLe64(record.time_us, out.subspan<detail::kAtTime, 8U>());
129 StoreLe16(record.source, out.subspan<detail::kAtSource, 2U>());
130 b[detail::kAtStatus] = static_cast<uint8_t>(record.status);
131 StoreLe32(record.detail, out.subspan<detail::kAtDetail, 4U>());
132 StoreLe32(record.value, out.subspan<detail::kAtValue, 4U>());
133 const std::span<const uint8_t> covered = std::span<const uint8_t>{b}.first(detail::kAtCrc);
134 StoreLe32(Crc32Ieee::Compute(covered), out.subspan<detail::kAtCrc, 4U>());
135 return b;
136}

144[[nodiscard]] constexpr kb::Result<FaultRecord> ParseRecord(
145 std::span<const uint8_t, kFaultRecordBytes> bytes) noexcept {
146 kb::Result<FaultRecord> result = kb::Result<FaultRecord>::Failure(kb::Status::Empty);
147 const uint8_t kind = bytes[detail::kAtKind];
148 const uint8_t status = bytes[detail::kAtStatus];
149 if (LoadLe16(bytes.first<2U>()) != detail::kJournalMagic) {
150 // Empty.
151 } else if (LoadLe32(bytes.subspan<detail::kAtCrc, 4U>()) !=
152 Crc32Ieee::Compute(bytes.first(detail::kAtCrc))) {
153 result = kb::Result<FaultRecord>::Failure(kb::Status::CrcMismatch);
154 } else if ((bytes[detail::kAtVersion] != detail::kJournalVersion) ||
155 (bytes[detail::kAtReserved] != 0U) || (kind == 0U) || (kind > detail::kLastKind) ||
156 (status >= kStatusCount)) {
157 result = kb::Result<FaultRecord>::Failure(kb::Status::Unsupported);
158 } else {
159 FaultRecord r{};
160 r.kind = static_cast<RecordKind>(kind);
161 r.sequence = LoadLe32(bytes.subspan<detail::kAtSequence, 4U>());
162 r.time_us = LoadLe64(bytes.subspan<detail::kAtTime, 8U>());
163 r.source = LoadLe16(bytes.subspan<detail::kAtSource, 2U>());
164 r.status = static_cast<kb::Status>(status);
165 r.detail = LoadLe32(bytes.subspan<detail::kAtDetail, 4U>());
166 r.value = LoadLe32(bytes.subspan<detail::kAtValue, 4U>());
167 result = kb::Result<FaultRecord>::Success(r);
168 }
169 return result;
170}

201 [[nodiscard]] constexpr kb::Status Read(std::size_t offset, std::span<uint8_t> out) noexcept {
202 const bool fits = fitsAt(offset, out.size());
203 if (fits) {
204 for (std::size_t i = 0U; i < out.size(); ++i) {
205 out[i] = bytes_[offset + i];

208 return fits ? kb::Status::Ok : kb::Status::OutOfRange;

213 const bool fits = fitsAt(offset, in.size());
214 if (fits) {
215 for (std::size_t i = 0U; i < in.size(); ++i) {
216 bytes_[offset + i] = in[i];
217 }
218 }
219 return fits ? kb::Status::Ok : kb::Status::OutOfRange;
220 }
222 [[nodiscard]] constexpr std::span<uint8_t, Bytes> Raw() noexcept {
223 return std::span<uint8_t, Bytes>{bytes_};

266 [[nodiscard]] kb::Result<JournalScan> Open() noexcept {
267 kb::Result<JournalScan> result = kb::Result<JournalScan>::Failure(kb::Status::OutOfRange);
268 if (store_.Capacity() >= (Slots * kFaultRecordBytes)) {
269 JournalScan scan{};
270 std::size_t newest_slot = Slots - 1U; // so that an empty journal starts at slot 0
271 kb::Status failed = kb::Status::Ok;
272 for (std::size_t i = 0U; (i < Slots) && (failed == kb::Status::Ok); ++i) {
273 FaultRecordBytes raw{};
274 failed = store_.Read(i * kFaultRecordBytes, raw);
275 const kb::Result<FaultRecord> r = ParseRecord(raw);
276 if (failed != kb::Status::Ok) {
277 // Reported below.
278 } else if (r.IsOk()) {
279 if ((scan.valid == 0U) || detail::SequenceNewer(r.Value().sequence, scan.newest)) {
280 scan.newest = r.Value().sequence;
281 newest_slot = i;
282 }
283 ++scan.valid;
284 } else {
285 count(r.Error(), scan);
286 }
287 }
288 if (failed == kb::Status::Ok) {
289 head_ = (newest_slot + 1U) % Slots;
290 next_ = (scan.valid == 0U) ? 1U : (scan.newest + 1U);
291 count_ = scan.valid;
292 open_ = true;
293 result = kb::Result<JournalScan>::Success(scan);
294 } else {
295 result = kb::Result<JournalScan>::Failure(failed);
296 }
297 }
298 return result;
299 }

305 [[nodiscard]] kb::Result<uint32_t> Append(FaultRecord record) noexcept {
306 kb::Result<uint32_t> result = kb::Result<uint32_t>::Failure(kb::Status::NotReady);
307 if (open_) {
308 record.sequence = next_;
309 const FaultRecordBytes raw = EncodeRecord(record);
310 const kb::Status s = store_.Write(head_ * kFaultRecordBytes, raw);
311 if (s == kb::Status::Ok) {
312 head_ = (head_ + 1U) % Slots;
313 ++next_;
314 count_ = (count_ < Slots) ? (count_ + 1U) : Slots;
315 ++stats_.appends;
316 result = kb::Result<uint32_t>::Success(record.sequence);
317 } else {
318 ++stats_.write_failures;
319 result = kb::Result<uint32_t>::Failure(s);
320 }
321 }
322 return result;
323 }

341 [[nodiscard]] kb::Result<FaultRecord> Read(std::size_t age) noexcept {
342 kb::Result<FaultRecord> result = kb::Result<FaultRecord>::Failure(kb::Status::NotReady);
343 if (!open_) {
344 // NotReady.
345 } else if (age >= count_) {
346 result = kb::Result<FaultRecord>::Failure(kb::Status::OutOfRange);
347 } else {
348 const std::size_t slot = ((head_ + Slots) - 1U - age) % Slots;
349 FaultRecordBytes raw{};
350 const kb::Status s = store_.Read(slot * kFaultRecordBytes, raw);
351 result = (s == kb::Status::Ok) ? ParseRecord(raw) : kb::Result<FaultRecord>::Failure(s);
352 }
353 return result;

357 [[nodiscard]] kb::Result<FaultRecord> FindNewest(RecordKind kind) noexcept {
358 kb::Result<FaultRecord> result =
359 kb::Result<FaultRecord>::Failure(open_ ? kb::Status::Empty : kb::Status::NotReady);
360 bool found = false;
361 for (std::size_t age = 0U; (age < count_) && (!found); ++age) {
362 const kb::Result<FaultRecord> r = Read(age);
363 if (r.IsOk() && (r.Value().kind == kind)) {
364 result = r;
365 found = true;
366 }
367 }
368 return result;
