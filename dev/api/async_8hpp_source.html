92 [[nodiscard]] constexpr kb::Result<std::size_t> Completion() const noexcept {
93 kb::Result<std::size_t> r = kb::Result<std::size_t>::Failure(kb::Status::NotReady);

232 [[nodiscard]] AsyncPoll Poll(kb::Instant now) noexcept {
233 static_cast<void>(now); // the transfer already ended in Start()
234 AsyncPoll p = AsyncPoll::InState(AsyncState::Idle);
235 if (done_) {
236 done_ = false;
237 p = result_;
238 }
239 return p;
240 }
242 [[nodiscard]] kb::Status Cancel() noexcept {
243 done_ = false;
244 return kb::Status::Ok;
245 }

248private:
249 template <typename Run>
250 [[nodiscard]] kb::Status start(const Run& run, std::size_t count,
251 kb::utils::Deadline deadline) noexcept {
252 kb::Status status = kb::Status::Busy;
253 if (!done_) {
254 status = kb::Status::Timeout;
255 if (!deadline.Expired(clock_.Now())) {
256 const kb::Status outcome = run();
257 const bool late = deadline.Expired(clock_.Now());
258 result_ = AsyncPoll::Completed(late ? kb::Status::Timeout : outcome, count);
259 done_ = true;
260 status = kb::Status::Ok;
261 }
262 }
263 return status;
264 }
